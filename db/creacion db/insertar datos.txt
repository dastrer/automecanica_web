DELIMITER //
CREATE TRIGGER actualizar_inventario_ingreso
AFTER INSERT ON articulo_ingreso
FOR EACH ROW
BEGIN
    UPDATE articulo
    SET Cantidad = Cantidad + NEW.Cantidad_ingreso
    WHERE Idarticulo = NEW.Idarticulo;
END;
//
DELIMITER ;

DELIMITER //
CREATE TRIGGER trg_actualizar_inventario_servicio
AFTER INSERT ON servicio_articulo
FOR EACH ROW
BEGIN
    UPDATE articulo
    SET Cantidad = Cantidad - NEW.Cantidad_venta
    WHERE Idarticulo = NEW.Idarticulo;
END;
//
DELIMITER ;


DELIMITER //
CREATE TRIGGER trg_no_servicio_sin_stock
BEFORE INSERT ON servicio_articulo
FOR EACH ROW
BEGIN
    DECLARE stock_actual INT;

    SELECT Cantidad INTO stock_actual
    FROM articulo
    WHERE Idarticulo = NEW.Idarticulo;

    IF stock_actual < NEW.Cantidad_venta THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Stock insuficiente para aplicar el artículo en el servicio';
    END IF;
END;
//
DELIMITER ;



DELIMITER //
CREATE TRIGGER actualizar_inventario_venta
AFTER INSERT ON articulo_venta
FOR EACH ROW
BEGIN
    UPDATE articulo
    SET Cantidad = Cantidad - NEW.Cantidad_venta
    WHERE Idarticulo = NEW.Idarticulo;
END;
//
DELIMITER ;
